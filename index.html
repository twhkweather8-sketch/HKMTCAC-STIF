            // --- WIM Drawer ---

            document.getElementById('pub-wim-wind').textContent = data.wimWind || '--';

            document.getElementById('pub-wim-rain').textContent = data.wimRain || '--';

            document.getElementById('pub-wim-action').innerHTML = `<li>${data.wimAction || '--'}</li>`;


            // --- WIM Grid Active Cell ---

            const pubGrid = document.getElementById('pub-wim-grid');

            if (pubGrid.children.length > 0) {

                pubGrid.querySelectorAll('.cell').forEach(c => {

                    c.classList.remove('active');

                    if (parseInt(c.dataset.x) === impact && parseInt(c.dataset.y) === (7 - certainty)) {

                        c.classList.add('active');

                    }

                });

            }


            // --- WIM Badge Color ---

            const wimBadge = document.getElementById('pub-wim-badge');

            if (impact <= 1) wimBadge.style.background = 'var(--green)';

            else if (impact <= 3) wimBadge.style.background = 'var(--yellow)';

            else if (impact <= 5) wimBadge.style.background = 'var(--orange)';

            else wimBadge.style.background = 'var(--red)';


        } catch (err) {

            console.error('Failed to load dashboard data:', err);

        }

    }


    // ============================================================

    // GENERATE 8x8 WIM MATRIX (Public View)

    // ============================================================

    function generateWIMGrid() {

        const grid = document.getElementById('pub-wim-grid');

        if (!grid) return;

        grid.innerHTML = '';


        const yLabels = ["極高 95-99%", "較高 90-95%", "高 80-90%", "中高 60-80%", "中等 40-60%", "中低 20-40%", "稍低 10-20%", "較低 0-10%"];

        const xLabels = ["較低", "稍低", "中低", "中等-中高", "中高", "高", "較高", "極高"];


        grid.appendChild(document.createElement('div'));

        xLabels.forEach(label => {

            const div = document.createElement('div');

            div.className = 'label-x';

            div.textContent = label;

            grid.appendChild(div);

        });


        for (let y = 0; y < 8; y++) {

            const yDiv = document.createElement('div');

            yDiv.className = 'label-y';

            yDiv.textContent = yLabels[y];

            grid.appendChild(yDiv);


            for (let x = 0; x < 8; x++) {

                const cell = document.createElement('div');

                cell.className = 'cell';

                cell.dataset.x = x;

                cell.dataset.y = y;


                const score = (7 - y) + x;

                if (score <= 2) cell.classList.add('c-grey');

                else if (score <= 4) cell.classList.add('c-blue');

                else if (score <= 6) cell.classList.add('c-green');

                else if (score <= 8) cell.classList.add('c-yellow');

                else if (score <= 10) cell.classList.add('c-orange');

                else if (score <= 12) cell.classList.add('c-red');

                else cell.classList.add('c-dark');


                grid.appendChild(cell);

            }

        }

    }

        const content = document.getElementById('pub-wim-content');

        const icon = document.getElementById('pub-wim-icon');

        content.classList.toggle('active');

        icon.textContent = content.classList.contains('active') ? '➖' : '➕';

    }
